```ts
export async function completePersonalOnboarding(
  classId: string,
  orgId: string,
  userId: string,
  data: CompletePersonalOnboardingDataOutput,
) {
  const result = await tryConstraint(
    prisma.$transaction(async (tx) => {
      await tx.userOrganization.update({
        where: { userId_orgId: { userId, orgId }, status: 'PENDING' },
        data: { status: 'ACTIVE' },
        select: { id: true },
      })

      return tx.class.update({
        where: { id: classId, programTrack: { orgId } },
        data,
        select: { id: true, name: true, type: true },
      })
    }),
  )

  await Promise.all([
    invalidateEvent('CLASS_UPDATED', orgId, classId),
    invalidateEvent('USER_ORGANIZATION_UPDATED', orgId),
  ])
  return result
}
```

**`databases/classe.queries.ts`**
```ts
export async function getPersonalClass(orgId: string) {
  'use cache'
  cacheTag(CACHE.CLASSES(orgId))
  cacheLife('hours')

  return prisma.class.findFirst({
    where: { programTrack: { orgId }, deletedAt: null },
    orderBy: { createdAt: 'asc' },
    select: { id: true, name: true, type: true },
  })
}
```
La query ne regarde plus le statut : c'est le layout qui décide `PENDING` ou non, et la query ne fait que retrouver la classe unique de l'espace.

**`actions/classe.mutations.ts`**
```ts
'use server'
import * as v from 'valibot'
import { authAccess } from '@/services/auth'
import { completePersonalOnboardingSchema } from '../validation'
import type { CompletePersonalOnboardingInput } from '../validation'
import { completePersonalOnboarding } from '../databases'

export async function completePersonalOnboardingAction(input: CompletePersonalOnboardingInput) {
  const auth = await authAccess({ requiredRole: 'TEACHER', requiredOrgType: 'PERSONAL' })
  if (!auth.data) return { error: auth.error }
  const { user, orgId } = auth.data

  const parsed = v.safeParse(completePersonalOnboardingSchema, input)
  if (!parsed.success) return { error: parsed.issues[0]?.message ?? 'Données invalides' }

  const result = await completePersonalOnboarding(
    parsed.output.classId,
    orgId,
    user.id,
    parsed.output.data,
  )
  return { data: result }
}
```

**`actions/classe.queries.ts`**
```ts
'use server'
import { authAccess } from '@/services/auth'
import { ERRORS } from '@/config'
import { getPersonalClass } from '../databases'

export async function getPersonalClassAction() {
  const auth = await authAccess({ requiredRole: 'TEACHER', requiredOrgType: 'PERSONAL' })
  if (!auth.data) return { error: auth.error }

  try {
    return { data: await getPersonalClass(auth.data.orgId) }
  } catch (e) {
    return { error: e instanceof Error ? e.message : ERRORS.SERVER }
  }
}
```

Deux points à tester, que le skill ne peut pas trancher. D'abord, `CACHE.CLASSES` est le nom que j'ai déduit du pattern `CACHE.<KEY>(orgId)`, donc à aligner sur ta clé réelle. Ensuite, si `authAccess` lit le statut dans `app_metadata` (c'est ce que dit ton schéma de métadonnées), passer à `ACTIVE` en base ne suffit pas : l'action doit aussi resynchroniser le statut via ton `setUserAccess`. Sinon le layout continuera de voir `PENDING` jusqu'au prochain refresh du token. Dis-moi si tu veux que j'ajoute cet appel.
