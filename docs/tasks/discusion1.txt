//crates\chat\relay\src\main.rs
use std::collections::{HashMap, HashSet};
use std::fmt::Display;
use std::process::exit;

use chat_relay::{start, HubConfig};

const USAGE: &str = "usage: chat-relay --token <jeton> [--port N] [--bind IP] [--members fichier.json] [--host-user ID] [--rate-burst N] [--rate-per-sec F] [--stop-on-stdin-eof]";

/// Toute option absente de cette liste est refusée : un ancien binaire ou une faute de frappe
/// ne doit jamais être ignoré en silence.
const KNOWN_OPTIONS: &[&str] = &[
    "--token",
    "--port",
    "--bind",
    "--members",
    "--host-user",
    "--rate-burst",
    "--rate-per-sec",
    "--stop-on-stdin-eof",
];
/// Options sans valeur.
const FLAGS: &[&str] = &["--stop-on-stdin-eof"];

struct Opts {
    values: HashMap<&'static str, String>,
    stop_on_stdin_eof: bool,
}

fn parse(args: &[String]) -> Result<Opts, String> {
    let mut values = HashMap::new();
    let mut stop_on_stdin_eof = false;
    let mut it = args.iter();
    while let Some(a) = it.next() {
        let Some(name) = KNOWN_OPTIONS.iter().find(|k| **k == a.as_str()) else {
            return Err(if a.starts_with("--") {
                format!("option inconnue : {a}")
            } else {
                format!("argument inattendu : {a}")
            });
        };
        if FLAGS.contains(name) {
            stop_on_stdin_eof = true;
            continue;
        }
        let Some(v) = it.next() else {
            return Err(format!("valeur manquante pour {name}"));
        };
        values.insert(*name, v.clone());
    }
    Ok(Opts { values, stop_on_stdin_eof })
}

/// Erreur d'utilisation : code de sortie 2, ligne `ERROR <code>: <message>`.
fn bad(msg: impl Display) -> ! {
    eprintln!("ERROR bad_options: {msg}");
    exit(2)
}

#[tokio::main]
async fn main() {
    let args: Vec<String> = std::env::args().skip(1).collect();
    let opts = parse(&args).unwrap_or_else(|e| bad(e));
    let get = |name: &str| opts.values.get(name).cloned();

    let Some(token) = get("--token") else {
        eprintln!("{USAGE}");
        exit(2);
    };
    if token.is_empty() {
        bad("le jeton est vide");
    }

    let bind = get("--bind").unwrap_or_else(|| "0.0.0.0".into());
    let port: u16 = match get("--port") {
        None => 0,
        Some(p) => p.parse().unwrap_or_else(|_| bad(format!("port illisible : {p}"))),
    };

    let mut cfg = HubConfig::new(token);
    cfg.host_user_id = get("--host-user");
    // Réglages du débit : outils de test et de développement
    if let Some(v) = get("--rate-burst") {
        match v.parse::<u32>() {
            Ok(n) if n >= 1 => cfg.rate_burst = n,
            _ => bad(format!("--rate-burst doit être un entier >= 1 : {v}")),
        }
    }
    if let Some(v) = get("--rate-per-sec") {
        match v.parse::<f64>() {
            Ok(f) if f.is_finite() && f > 0.0 => cfg.rate_per_sec = f,
            _ => bad(format!("--rate-per-sec doit être un nombre > 0 : {v}")),
        }
    }
    if let Some(path) = get("--members") {
        let raw = std::fs::read_to_string(&path)
            .unwrap_or_else(|e| bad(format!("--members illisible ({path}) : {e}")));
        let parsed: HashMap<String, Vec<String>> = serde_json::from_str(&raw)
            .unwrap_or_else(|_| bad("--members : JSON {canal: [utilisateurs]} attendu"));
        cfg.members = Some(
            parsed
                .into_iter()
                .map(|(k, v)| (k, v.into_iter().collect::<HashSet<_>>()))
                .collect(),
        );
    }

    let handle = match start(cfg, &format!("{bind}:{port}")).await {
        Ok(handle) => handle,
        Err(e) => {
            // `code` est stable (lu par les hôtes Node et Tauri) ; le message est pour l'humain
            eprintln!("ERROR {}: {e}", e.code());
            exit(1);
        }
    };
    println!("LISTENING {}", handle.addr);

    // Arrêt par fermeture de stdin : seul signal propre sous Windows. Optionnel, car un relais
    // lancé sans terminal verrait sinon un EOF immédiat et s'arrêterait tout seul.
    let stdin_closed = if opts.stop_on_stdin_eof {
        let (tx, rx) = tokio::sync::oneshot::channel::<()>();
        // Fil dédié : une lecture bloquante ne doit pas retenir l'arrêt du runtime
        std::thread::spawn(move || {
            let _ = std::io::copy(&mut std::io::stdin().lock(), &mut std::io::sink());
            let _ = tx.send(());
        });
        Some(rx)
    } else {
        None
    };

    tokio::select! {
        _ = tokio::signal::ctrl_c() => {}
        _ = async {
            match stdin_closed {
                Some(rx) => { let _ = rx.await; }
                None => std::future::pending::<()>().await,
            }
        } => {}
    }
    handle.stop().await;
}

const BIN: &str = env!("CARGO_BIN_EXE_chat-relay");

fn spawn(args: &[&str]) -> Child {
    Command::new(BIN)
        .args(args)
        .stdin(Stdio::piped())
        .stdout(Stdio::piped())
        .stderr(Stdio::piped())
        .spawn()
        .expect("lancement du binaire")
}

/// Lit stdout jusqu'à `LISTENING <adresse>` et rend l'adresse.
fn listening(child: &mut Child) -> String {
    let mut lines = BufReader::new(child.stdout.take().unwrap()).lines();
    let line = lines.next().expect("stdout fermé").expect("lecture");
    line.strip_prefix("LISTENING ").expect("ligne LISTENING").to_string()
}

fn wait_exit(child: &mut Child, within: Duration) -> Option<std::process::ExitStatus> {
    let end = Instant::now() + within;
    while Instant::now() < end {
        if let Some(status) = child.try_wait().unwrap() {
            return Some(status);
        }
        std::thread::sleep(Duration::from_millis(25));
    }
    None
}

fn stderr_of(mut child: Child) -> String {
    let mut s = String::new();
    child.stderr.take().unwrap().read_to_string(&mut s).unwrap();
    s
}

/// Lance, attend une sortie rapide et rend (code, stderr).
fn fails(args: &[&str]) -> (Option<i32>, String) {
    let mut c = spawn(args);
    let status = wait_exit(&mut c, Duration::from_secs(5)).unwrap_or_else(|| {
        let _ = c.kill();
        panic!("le binaire devait sortir rapidement")
    });
    (status.code(), stderr_of(c))
}

const USAGE: &str = "usage: chat-relay --token <jeton> [--port N] [--bind IP] [--members fichier.json] [--host-user ID] [--rate-burst N] [--rate-per-sec F] [--stop-on-stdin-eof]";

/// Toute option absente de cette liste est refusée : un ancien binaire ou une faute de frappe
/// ne doit jamais être ignoré en silence.
const KNOWN_OPTIONS: &[&str] = &[
    "--token",
    "--port",
    "--bind",
    "--members",
    "--host-user",
    "--rate-burst",
    "--rate-per-sec",
    "--stop-on-stdin-eof",
];
/// Options sans valeur.
const FLAGS: &[&str] = &["--stop-on-stdin-eof"];

struct Opts {
    values: HashMap<&'static str, String>,
    stop_on_stdin_eof: bool,
}

fn parse(args: &[String]) -> Result<Opts, String> {
    let mut values = HashMap::new();
    let mut stop_on_stdin_eof = false;
    let mut it = args.iter();
    while let Some(a) = it.next() {
        let Some(name) = KNOWN_OPTIONS.iter().find(|k| **k == a.as_str()) else {
            return Err(if a.starts_with("--") {
                format!("option inconnue : {a}")
            } else {
                format!("argument inattendu : {a}")
            });

            stop_on_stdin_eof = true;
            continue;
        }
        let Some(v) = it.next() else {
            return Err(format!("valeur manquante pour {name}"));
        };
        values.insert(*name, v.clone());
    }
    Ok(Opts { values, stop_on_stdin_eof })
}

    eprintln!("ERROR bad_options: {msg}");

}

            exit(1);

    // lancé sans terminal verrait sinon un EOF immédiat et s'arrêterait tout seul.

            let _ = tx.send(());
        });
        Some(rx)
    } else {
        None
    };

        _ = tokio::signal::ctrl_c() => {}

                Some(rx) => { let _ = rx.await; }
                None => std::future::pending::<()>().await,
            }
        } => {}

const BIN: &str = env!("CARGO_BIN_EXE_chat-relay");

fn spawn(args: &[&str]) -> Child {
    Command::new(BIN)
        .args(args)
        .stdin(Stdio::piped())
        .stdout(Stdio::piped())
        .stderr(Stdio::piped())
        .spawn()
        .expect("lancement du binaire")

/// Lit stdout jusqu'à `LISTENING <adresse>` et rend l'adresse.
fn listening(child: &mut Child) -> String {
    let mut lines = BufReader::new(child.stdout.take().unwrap()).lines();
    let line = lines.next().expect("stdout fermé").expect("lecture");
    line.strip_prefix("LISTENING ").expect("ligne LISTENING").to_string()
}

fn wait_exit(child: &mut Child, within: Duration) -> Option<std::process::ExitStatus> {
    let end = Instant::now() + within;
    while Instant::now() < end {
        if let Some(status) = child.try_wait().unwrap() {
            return Some(status);
        }
        std::thread::sleep(Duration::from_millis(25));
    }
    None
}

fn stderr_of(mut child: Child) -> String {
    let mut s = String::new();
    child.stderr.take().unwrap().read_to_string(&mut s).unwrap();
    s
}

/// Lance, attend une sortie rapide et rend (code, stderr).

        panic!("le binaire devait sortir rapidement")

}

Conception
Inactivité : un Sleep repoussé à chaque trame reçue. À l'échéance, le relais envoie un Close (code 1008, raison idle_timeout) puis ferme. Seules les trames reçues comptent, pas celles qu'on envoie. Elle ne s'applique qu'après le hello, qui a déjà son délai.
File bornée : mpsc::channel(capacité) et try_send. Sur Full, le client est retiré de clients immédiatement, ce qui borne la mémoire et corrige peer_count. Un signal Notify réveille sa tâche pour qu'elle envoie un Close (code 1013, raison slow_consumer). Il est traité en priorité par rapport à la file (biased). Les messages déjà en file sont abandonnés. Le client rattrape par known à sa reconnexion.
Délai d'écriture : une fonction qui enveloppe chaque sink.send et sink.close dans un timeout. Si le délai est dépassé, la tâche s'arrête et le socket est abandonné.
Écart avec ce que tu avais décidé

Tu avais écrit « file pleine = on ferme le client lent par une trame Close ». Ce n'est garanti que si le socket du client accepte encore une écriture. Quand ses tampons TCP sont pleins, le Close ne peut pas partir. Ce client est alors déconnecté par le délai d'écriture, sans Close.

Je documente donc le Close comme au mieux (best-effort).
La garantie qui compte est que le client soit retiré, que la mémoire soit bornée et que la tâche soit libérée.
